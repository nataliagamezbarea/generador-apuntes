\subtituloA{Rutas a archivos JavaScript}

\lista{
  \textbf{Ruta relativa:} Se dirige desde el directorio actual del documento HTML (ejemplo: \cmd{../js/ejemplo.js}). |
  \textbf{Ruta absoluta:} Se dirige desde la raíz del sistema de archivos o unidad. |
  \textbf{Ruta remota:} Utiliza un protocolo como \cmd{http} (ejemplo: \cmd{http://www.tu-dominio.com/ejemplo.js}).
}
