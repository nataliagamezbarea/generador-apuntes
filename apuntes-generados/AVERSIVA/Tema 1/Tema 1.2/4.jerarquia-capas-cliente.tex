\subtituloA{Jerarquía de capas en el cliente}

\lista{
  \textbf{Capa de contenido:} Texto, imágenes y elementos brutos del documento HTML. |
  \textbf{Capa de estructura:} Definida por el Document Object Model (DOM) y la estructura HTML. |
  \textbf{Capa de presentación:} Definida por CSS, controla el estilo visual. |
  \textbf{Capa de comportamiento:} Gestionada por JavaScript, controla la interactividad y la lógica del cliente.
}
