\subtituloA{Modelo Cliente-Servidor}

\definicion{Modelo Cliente-Servidor}{Arquitectura web predominante donde la máquina cliente (navegador) solicita recursos al servidor, el cual los procesa y los entrega.}

\lista{
  \item{Servidor: Aloja hardware, software, técnicas y lenguajes (se aconseja soporte multihilos).} \item{Máquina Cliente: Ejecuta peticiones mediante navegadores que deben soportar HTML, CSS, JavaScript y controles ActiveX.} \item{Conectividad: La red y sus características determinan la comunicación efectiva.}
}

\nota{Para el examen}{
¿Qué componente es responsable de realizar la petición de solicitud web al servidor?

\textbf{Respuesta:} La máquina cliente (a través del navegador).
}
